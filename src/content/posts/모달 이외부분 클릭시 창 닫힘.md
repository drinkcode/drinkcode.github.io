# 모달 이외부분 클릭시 창 닫힘 코드

```js
import React, { useEffect } from "react"
import toothImg from "../../images/artificial.png"

export default ({ setModalOnOff }) => {
  useEffect(() => {
    var model_wrap = document.getElementById("model_wrap")
    var modal = document.getElementById("modal")

    model_wrap.addEventListener(
      "click",
      function () {
        setModalOnOff(false)
        // alert('wrapper');
      },
      false
    )
    modal.addEventListener(
      "click",
      function (ev) {
        ev.stopPropagation() // 이 부분이 가장 중요함, 이걸로 인해서 모달 이외의 부분만 클릭했을 때 모달 창이 꺼지는 효과를 만들 수 있음
      },
      false
    )
  }, [])
  return (
    <div
      id="model_wrap"
      style={{
        display: "table",
        backgroundColor: "rgba(0,0,0,0.7)",
        width: "100vw",
        height: "100vh",
        position: "absolute",
        zIndex: 900,
      }}
    >
      <div
        id="modal"
        style={{
          backgroundColor: "#181818",
          position: "absolute",
          top: "50%",
          left: "50%",
          width: "70vw",
          height: "70vh",
          marginTop: "-35vh",
          marginLeft: "-35vw",
          color: "white",
          zIndex: 9999,
        }}
      >
        <h1 style={{ fontSize: "80px" }}>Select Image</h1>
        <img src={toothImg} width="400px" height="300px" />
      </div>
    </div>
  )
}
```
